        <div v-if="toast.show" class="fixed bottom-5 right-5 z-50 px-4 py-3 rounded-xl shadow-lg text-white text-xs font-medium flex items-center gap-2 transition-all duration-300" :class="toast.type === 'success' ? 'bg-emerald-600' : 'bg-rose-600'">
            <i :class="toast.type === 'success' ? 'fa-solid fa-circle-check' : 'fa-solid fa-circle-exclamation'"></i>
            <span>{{ toast.message }}</span>
        </div>

        <!-- Footer -->
        <footer class="bg-slate-800 text-slate-400 py-4 text-center text-xs border-t border-slate-700 mt-auto">
            ศูนย์แพทยศาสตรศึกษาชั้นคลินิก โรงพยาบาลมหาราชนครราชสีมา &copy; 2026 - All Rights Reserved.
        </footer>

    </div>

    <script>
        const { createApp, ref, computed, onMounted } = Vue;

        createApp({
            setup() {
                // Application State
                const currentTab = ref('form');
                const isAdmin = ref(false);
                const showPreviewModal = ref(false);
                const isPreviewModeSubmission = ref(false);
                const showPinModal = ref(false);
                const isChangingPin = ref(false);
                const showApproverModal = ref(false);
                
                const pinInput = ref('');
                const newPinInput = ref('');
                const storedPin = ref('1234'); // Default PIN
                const searchQuery = ref('');

                // Approvers List & Form State
                const defaultApproversList = [
                    { id: '1', name: 'ศ.ดร.นพ. สมชาย รักดี', position: 'ประธานคณะกรรมการวิจัย ศศค.', isDefault: true },
                    { id: '2', name: 'ผศ.ดร.พญ. วิภาวี มีสุข', position: 'ผู้อำนวยการศูนย์แพทยศาสตรศึกษาชั้นคลินิก', isDefault: false }
                ];
                const approversList = ref([]);
                const newApprover = ref({ name: '', position: '' });

                // Notification Toast
                const toast = ref({ show: false, message: '', type: 'success' });

                const showToast = (msg, type = 'success') => {
                    toast.value = { show: true, message: msg, type };
                    setTimeout(() => toast.value.show = false, 3500);
                };

                // Helper: Generate Auto Receipt Number in format YY/MM-XXX
                const generateAutoReceiptNo = () => {
                    const now = new Date();
                    const thaiYearTwoDigits = String(now.getFullYear() + 543).slice(-2);
                    const monthTwoDigits = String(now.getMonth() + 1).padStart(2, '0');
                    const seqNumber = String(records.value.length + 1).padStart(3, '0');
                    return `${thaiYearTwoDigits}/${monthTwoDigits}-${seqNumber}`;
                };

                // Helper: Today formatted string
                const getTodayFormatted = () => {
                    const now = new Date();
                    const day = String(now.getDate()).padStart(2, '0');
                    const month = String(now.getMonth() + 1).padStart(2, '0');
                    const year = now.getFullYear() + 543;
                    return `${day}/${month}/${year}`;
                };

                // Get Default Approver from list
                const getDefaultApprover = () => {
                    const foundDefault = approversList.value.find(a => a.isDefault);
                    if (foundDefault) return foundDefault;
                    return approversList.value.length > 0 ? approversList.value[0] : { name: '', position: '' };
                };

                // Form Reactive Data Structure
                const createEmptyFormData = () => {
                    const defApprover = getDefaultApprover();
                    return {
                        id: Date.now().toString(),
                        fiscal_year: new Date().getFullYear() + 543,
                        created_at: getTodayFormatted(),
                        notifier_name: '',
                        position_group: '',
                        phone_email: '',
                        author_status: 'ผู้วิจัยหลัก',
                        title_th: '',
                        title_en: '',
                        work_type: { med_ed: false, clinic: false, innovation: false },
                        work_status: 'วิจัยสำเร็จ นำเสนอ',
                        presentation_type: 'Oral',
                        keywords: '',
                        authors: [{ name: '', affiliation: '', percentage: 100 }],
                        presenter_name: '',
                        conference_name_place: '',
                        conference_format: 'Onsite',
                        abstract_link: '',
                        journal_name: '',
                        article_doi_link: '',
                        work_year_ad: new Date().getFullYear(),
                        support_from_mec: { budget: false, personnel: false, none: true },
                        certifications: {
                            never_applied: false,
                            never_rewarded: false,
                            no_duplicate_author: false,
                            truthful: false
                        },
                        attached_files: {
                            abstract: null,
                            presentation: null,
                            publication: null,
                            other: null
                        },
                        attached_files_other_detail: '',
                        admin_review: {
                            receipt_no: '',
                            receipt_date: getTodayFormatted(),
                            check_complete: true,
                            check_traceable: true,
                            check_qualifications: true,
                            check_not_duplicate: true,
                            result: '',
                            reason: '',
                            passed: true,
                            approver_name: defApprover ? defApprover.name : '',
                            approver_position: defApprover ? defApprover.position : ''
                        }
                    };
                };

                const formData = ref(createEmptyFormData());
                const activePreviewData = ref(createEmptyFormData());
                const records = ref([]);

                // Computed total percentage
                const totalPercentage = computed(() => {
                    return formData.value.authors.reduce((sum, item) => sum + (Number(item.percentage) || 0), 0);
                });

                // Computed Dashboard Stats
                const dashboardStats = computed(() => {
                    const total = records.value.length;
                    const approved = records.value.filter(r => r.admin_review?.result === 'อนุมัติ').length;
                    const pending = records.value.filter(r => !r.admin_review?.result || r.admin_review?.result === 'รอดำเนินการ').length;
                    const revision = records.value.filter(r => r.admin_review?.result === 'ให้แก้ไข').length;
                    const rejected = records.value.filter(r => r.admin_review?.result === 'ไม่อนุมัติ').length;
                    const duplicate = records.value.filter(r => r.admin_review?.result === 'ข้อมูลซ้ำ').length;

                    const medEd = records.value.filter(r => r.work_type?.med_ed).length;
                    const clinic = records.value.filter(r => r.work_type?.clinic).length;
                    const innovation = records.value.filter(r => r.work_type?.innovation).length;

                    return { total, approved, pending, revision, rejected, duplicate, medEd, clinic, innovation };
                });

                // Filtered records for Admin & User List Table
                const filteredRecords = computed(() => {
                    if (!searchQuery.value) return records.value;
                    const q = searchQuery.value.toLowerCase();
                    return records.value.filter(r => 
                        (r.notifier_name && r.notifier_name.toLowerCase().includes(q)) ||
                        (r.title_th && r.title_th.toLowerCase().includes(q)) ||
                        (r.title_en && r.title_en.toLowerCase().includes(q)) ||
                        (r.position_group && r.position_group.toLowerCase().includes(q)) ||
                        (r.admin_review?.receipt_no && r.admin_review.receipt_no.toLowerCase().includes(q)) ||
                        (r.admin_review?.approver_name && r.admin_review.approver_name.toLowerCase().includes(q))
                    );
                });

                // Check Duplicate Title helper
                const isDuplicateTitle = (titleTh, titleEn, currentId = null) => {
                    const cleanTh = (titleTh || '').trim().toLowerCase();
                    const cleanEn = (titleEn || '').trim().toLowerCase();
                    if (!cleanTh && !cleanEn) return false;

                    return records.value.some(r => {
                        if (currentId && r.id === currentId) return false;
                        const existTh = (r.title_th || '').trim().toLowerCase();
                        const existEn = (r.title_en || '').trim().toLowerCase();

                        const matchTh = cleanTh && existTh && cleanTh === existTh;
                        const matchEn = cleanEn && existEn && cleanEn === existEn;
                        return matchTh || matchEn;
                    });
                };

                // Load Data on Mount
                onMounted(() => {
                    // Load Approvers
                    const savedApprovers = localStorage.getItem('mec_approvers_list');
                    if (savedApprovers) {
                        try { approversList.value = JSON.parse(savedApprovers); } catch(e) { approversList.value = defaultApproversList; }
                    } else {
                        approversList.value = defaultApproversList;
                        saveApproversToStorage();
                    }

                    // Load Records
                    const savedRecords = localStorage.getItem('mec_research_records');
                    if (savedRecords) {
                        try { records.value = JSON.parse(savedRecords); } catch (e) {}
                    }
                    const savedPin = localStorage.getItem('mec_admin_pin');
                    if (savedPin) storedPin.value = savedPin;

                    // Initialize formData
                    formData.value = createEmptyFormData();
                });

                const saveRecordsToStorage = () => {
                    localStorage.setItem('mec_research_records', JSON.stringify(records.value));
                };

                const saveApproversToStorage = () => {
                    localStorage.setItem('mec_approvers_list', JSON.stringify(approversList.value));
                };

                // Approvers Management Functions
                const addApprover = () => {
                    if (!newApprover.value.name.trim()) {
                        showToast('กรุณากรอกชื่อ-นามสกุล ผู้มีสิทธิ์อนุมัติ', 'error');
                        return;
                    }
                    const newItem = {
                        id: Date.now().toString(),
                        name: newApprover.value.name.trim(),
                        position: newApprover.value.position.trim() || 'คณะกรรมการวิจัย ศศค.',
                        isDefault: approversList.value.length === 0
                    };
                    approversList.value.push(newItem);
                    saveApproversToStorage();
                    newApprover.value = { name: '', position: '' };
                    showToast('เพิ่มผู้มีสิทธิ์อนุมัติสำเร็จ');
                };

                const deleteApprover = (id) => {
                    approversList.value = approversList.value.filter(a => a.id !== id);
                    if (approversList.value.length > 0 && !approversList.value.some(a => a.isDefault)) {
                        approversList.value[0].isDefault = true;
                    }
                    saveApproversToStorage();
                    showToast('ลบรายชื่อผู้มีสิทธิ์อนุมัติแล้ว');
                };

                const setDefaultApprover = (id) => {
                    approversList.value.forEach(a => {
                        a.isDefault = (a.id === id);
                    });
                    saveApproversToStorage();
                    showToast('ตั้งค่าผู้มีสิทธิ์อนุมัติเริ่มต้นสำเร็จ');
                };

                const onSelectApproverChange = (event) => {
                    const selectedId = event.target.value;
                    const found = approversList.value.find(a => a.id === selectedId);
                    if (found) {
                        activePreviewData.value.admin_review.approver_name = found.name;
                        activePreviewData.value.admin_review.approver_position = found.position;
                    }
                };

                // Author Row Management
                const addAuthor = () => {
                    formData.value.authors.push({ name: '', affiliation: '', percentage: 0 });
                };

                const removeAuthor = (index) => {
                    formData.value.authors.splice(index, 1);
                };

                // File Upload Base64 Handler
                const handleFileUpload = (event, category) => {
                    const file = event.target.files[0];
                    if (!file) return;

                    if (file.size > 10 * 1024 * 1024) {
                        showToast('ขนาดไฟล์เกิน 10MB กรุณาเลือกไฟล์ขนาดเล็กกว่านี้', 'error');
                        event.target.value = '';
                        return;
                    }

                    const reader = new FileReader();
                    reader.onload = (e) => {
                        formData.value.attached_files[category] = {
                            name: file.name,
                            type: file.type,
                            dataUrl: e.target.result
                        };
                        showToast(`แนบไฟล์ ${file.name} เรียบร้อยแล้ว`);
                    };
                    reader.readAsDataURL(file);
                };

                // Open/View Attached File
                const openFile = (fileObj) => {
                    if (!fileObj || !fileObj.dataUrl) {
                        showToast('ไม่พบข้อมูลไฟล์', 'error');
                        return;
                    }
                    const win = window.open();
                    if (win) {
                        win.document.write(`<iframe src="${fileObj.dataUrl}" frameborder="0" style="border:0; top:0px; left:0px; bottom:0px; right:0px; width:100%; height:100%;" allowfullscreen></iframe>`);
                    } else {
                        showToast('เบราว์เซอร์บล็อกการเปิดหน้าต่างใหม่', 'error');
                    }
                };

                // Trigger Preview Modal from Form
                const triggerPreviewModal = () => {
                    const activeCopy = JSON.parse(JSON.stringify(formData.value));
                    
                    // Generate Auto Receipt No YY/MM-XXX if not present
                    if (!activeCopy.admin_review.receipt_no) {
                        activeCopy.admin_review.receipt_no = generateAutoReceiptNo();
                    }
                    if (!activeCopy.admin_review.receipt_date) {
                        activeCopy.admin_review.receipt_date = getTodayFormatted();
                    }

                    // Populate default approver if not filled
                    if (!activeCopy.admin_review.approver_name) {
                        const def = getDefaultApprover();
                        activeCopy.admin_review.approver_name = def.name;
                        activeCopy.admin_review.approver_position = def.position;
                    }

                    activePreviewData.value = activeCopy;
                    isPreviewModeSubmission.value = true;
                    showPreviewModal.value = true;
                };

                // Confirm and Save Final Submission
                const confirmAndSaveSubmission = () => {
                    const newRecord = JSON.parse(JSON.stringify(activePreviewData.value));
                    newRecord.id = Date.now().toString();
                    newRecord.created_at = getTodayFormatted();

                    // Check for duplicate title
                    if (isDuplicateTitle(newRecord.title_th, newRecord.title_en)) {
                        newRecord.admin_review.result = 'ข้อมูลซ้ำ';
                        newRecord.admin_review.passed = false;
                        newRecord.admin_review.check_not_duplicate = false;
                        newRecord.admin_review.reason = 'ตรวจพบชื่อผลงานวิจัยซ้ำกับในระบบ';
                        showToast('ตรวจพบชื่อผลงานซ้ำในระบบ! บันทึกสถานะผลการพิจารณาเป็น "ข้อมูลซ้ำ"', 'error');
                    } else {
                        showToast('บันทึกข้อมูลผลงานวิจัยเข้าระบบเรียบร้อยแล้ว!');
                    }

                    records.value.unshift(newRecord);
                    saveRecordsToStorage();

                    showPreviewModal.value = false;
                    formData.value = createEmptyFormData();
                };

                // Admin Functions
                const openAdminAccess = () => {
                    if (isAdmin.value) {
                        currentTab.value = 'admin';
                    } else {
                        showPinModal.value = true;
                    }
                };

                const verifyAdminPin = () => {
                    if (pinInput.value === storedPin.value) {
                        isAdmin.value = true;
                        currentTab.value = 'admin';
                        showPinModal.value = false;
                        pinInput.value = '';
                        showToast('เข้าสู่ระบบผู้ดูแลเรียบร้อยแล้ว');
                    } else {
                        showToast('รหัสผ่าน PIN ไม่ถูกต้อง', 'error');
                    }
                };

                const logoutAdmin = () => {
                    isAdmin.value = false;
                    if (currentTab.value === 'admin') {
                        currentTab.value = 'user_list';
                    }
                    showToast('ออกจากระบบเจ้าหน้าที่เรียบร้อยแล้ว');
                };

                const changeAdminPin = () => {
                    if (pinInput.value !== storedPin.value) {
                        showToast('รหัส PIN ปัจจุบันไม่ถูกต้อง', 'error');
                        return;
                    }
                    if (!newPinInput.value || newPinInput.value.length < 4) {
                        showToast('รหัส PIN ใหม่ต้องมีอย่างน้อย 4 หลัก', 'error');
                        return;
                    }
                    storedPin.value = newPinInput.value;
                    localStorage.setItem('mec_admin_pin', newPinInput.value);
                    closePinModal();
                    showToast('เปลี่ยนรหัสผ่าน PIN สำเร็จ');
                };

                const closePinModal = () => {
                    showPinModal.value = false;
                    isChangingPin.value = false;
                    pinInput.value = '';
                    newPinInput.value = '';
                };

                const openItemDetail = (item) => {
                    const activeCopy = JSON.parse(JSON.stringify(item));
                    if (!activeCopy.admin_review.approver_name) {
                        const def = getDefaultApprover();
                        activeCopy.admin_review.approver_name = def.name;
                        activeCopy.admin_review.approver_position = def.position;
                    }
                    activePreviewData.value = activeCopy;
                    isPreviewModeSubmission.value = false;
                    showPreviewModal.value = true;
                };

                const saveAdminReview = () => {
                    const index = records.value.findIndex(r => r.id === activePreviewData.value.id);
                    if (index !== -1) {
                        records.value[index].admin_review = JSON.parse(JSON.stringify(activePreviewData.value.admin_review));
                        saveRecordsToStorage();
                        showToast('บันทึกผลการพิจารณา ส่วนที่ ๔ เรียบร้อยแล้ว');
                    }
                };

                const deleteRecord = (id) => {
                    if (window.confirm('คุณต้องการลบรายการนี้ใช่หรือไม่?')) {
                        records.value = records.value.filter(r => r.id !== id);
                        saveRecordsToStorage();
                        showToast('ลบรายการเรียบร้อยแล้ว');
                    }
                };

                // Data Backup JSON Export
                const exportBackupJSON = () => {
                    const backupObject = {
                        records: records.value,
                        approvers: approversList.value
                    };
                    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(backupObject, null, 2));
                    const downloadAnchor = document.createElement('a');
                    downloadAnchor.setAttribute("href", dataStr);
                    downloadAnchor.setAttribute("download", `mec_research_backup_${new Date().toISOString().slice(0,10)}.json`);
                    document.body.appendChild(downloadAnchor);
                    downloadAnchor.click();
                    downloadAnchor.remove();
                    showToast('ส่งออกไฟล์สำรองข้อมูล JSON สำเร็จ');
                };

                // Data Restore JSON Import
                const importRestoreJSON = (event) => {
                    const file = event.target.files[0];
                    if (!file) return;

                    const reader = new FileReader();
                    reader.onload = (e) => {
                        try {
                            const importedData = JSON.parse(e.target.result);
                            if (Array.isArray(importedData)) {
                                records.value = importedData;
                                saveRecordsToStorage();
                                showToast('เรียกคืนข้อมูลสำเร็จเรียบร้อย');
                            } else if (importedData.records && Array.isArray(importedData.records)) {
                                records.value = importedData.records;
                                if (importedData.approvers && Array.isArray(importedData.approvers)) {
                                    approversList.value = importedData.approvers;
                                    saveApproversToStorage();
                                }
                                saveRecordsToStorage();
                                showToast('เรียกคืนข้อมูลสำเร็จเรียบร้อย');
                            } else {
                                showToast('รูปแบบไฟล์ JSON ไม่ถูกต้อง', 'error');
                            }
                        } catch (err) {
                            showToast('เกิดข้อผิดพลาดในการอ่านไฟล์ JSON', 'error');
                        }
                    };
                    reader.readAsText(file);
                };

                // Export Table to Excel
                const exportToExcel = () => {
                    const exportData = records.value.map(r => ({
                        'เลขรับที่': r.admin_review?.receipt_no || '-',
                        'วันที่รับ': r.admin_review?.receipt_date || r.created_at,
                        'ชื่อผู้แจ้ง': r.notifier_name,
                        'ตำแหน่ง/กลุ่มงาน': r.position_group,
                        'โทรศัพท์/อีเมล': r.phone_email,
                        'ชื่อผลงาน (ไทย)': r.title_th,
                        'ชื่อผลงาน (อังกฤษ)': r.title_en || '-',
                        'สถานะผลงาน': r.work_status,
                        'การนำเสนอ': r.presentation_type,
                        'วารสารที่ตีพิมพ์': r.journal_name || '-',
                        'ปี ค.ศ.': r.work_year_ad,
                        'ผู้พิจารณาอนุมัติ': r.admin_review?.approver_name || '-',
                        'ตำแหน่งผู้พิจารณา': r.admin_review?.approver_position || '-',
                        'ผลการพิจารณา': r.admin_review?.result || 'รอดำเนินการ'
                    }));

                    const worksheet = XLSX.utils.json_to_sheet(exportData);
                    const workbook = XLSX.utils.book_new();
                    XLSX.utils.book_append_sheet(workbook, worksheet, "ข้อมูลงานวิจัย");
                    XLSX.writeFile(workbook, `รายการผลงานวิจัย_ศศค_${new Date().toISOString().slice(0,10)}.xlsx`);
                    showToast('ส่งออกไฟล์ Excel เรียบร้อยแล้ว');
                };

                // Reliable PDF Export Fix
                const exportToPDF = () => {
                    const element = document.getElementById('pdf-export-template');
                    if (!element) return;

                    showToast('กำลังประมวลผลไฟล์ PDF...');

                    const opt = {
                        margin:       [8, 8, 8, 8],
                        filename:     `แบบฟอร์มขอรับค่าตอบแทนวิจัย_${activePreviewData.value.notifier_name || 'doc'}.pdf`,
                        image:        { type: 'jpeg', quality: 0.98 },
                        html2canvas:  { scale: 2, useCORS: true, logging: false },
                        jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }
                    };

                    html2pdf().set(opt).from(element).save().then(() => {
                        showToast('ดาวน์โหลด PDF เรียบร้อยแล้ว');
                    }).catch(err => {
                        showToast('เกิดข้อผิดพลาดในการดาวน์โหลด PDF', 'error');
                        console.error(err);
                    });
                };

                return {
                    currentTab,
                    isAdmin,
                    showPreviewModal,
                    isPreviewModeSubmission,
                    showPinModal,
                    isChangingPin,
                    showApproverModal,
                    approversList,
                    newApprover,
                    pinInput,
                    newPinInput,
                    searchQuery,
                    toast,
                    formData,
                    activePreviewData,
                    records,
                    filteredRecords,
                    dashboardStats,
                    totalPercentage,
                    addAuthor,
                    removeAuthor,
                    handleFileUpload,
                    openFile,
                    triggerPreviewModal,
                    confirmAndSaveSubmission,
                    openAdminAccess,
                    verifyAdminPin,
                    logoutAdmin,
                    changeAdminPin,
                    closePinModal,
                    openItemDetail,
                    saveAdminReview,
                    deleteRecord,
                    addApprover,
                    deleteApprover,
                    setDefaultApprover,
                    onSelectApproverChange,
                    exportBackupJSON,
                    importRestoreJSON,
                    exportToExcel,
                    exportToPDF
                };
            }
        }).mount('#app');
    </script>
